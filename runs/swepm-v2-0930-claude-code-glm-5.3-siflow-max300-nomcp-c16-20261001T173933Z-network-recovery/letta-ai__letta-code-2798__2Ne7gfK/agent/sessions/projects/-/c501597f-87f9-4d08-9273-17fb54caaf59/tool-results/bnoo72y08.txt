     8	function normalizeExtensionConversationHistoryLimit(limit?: number): number {
     9	  if (limit === undefined) return DEFAULT_EXTENSION_CONVERSATION_HISTORY_LIMIT;
    10	  if (!Number.isFinite(limit)) {
    11	    return DEFAULT_EXTENSION_CONVERSATION_HISTORY_LIMIT;
    12	  }
    13	  return Math.min(
    14	    Math.max(1, Math.trunc(limit)),
    15	    MAX_EXTENSION_CONVERSATION_HISTORY_LIMIT,
    16	  );
    17	}
    18	
    19	export async function loadExtensionConversationHistoryFromBackend(
    20	  backend: Pick<Backend, "listConversationMessages">,
    21	  scope: {
    22	    agentId?: string | null;
    23	    conversationId?: string | null;
    24	  },
    25	  options?: ExtensionConversationHistoryOptions,
    26	): Promise<Message[]> {
    27	  const conversationId = scope.conversationId ?? "default";
    28	  const agentId = scope.agentId ?? null;
    29	  if (!scope.conversationId && !agentId) {
    30	    return [];
    31	  }
    32	
    33	  const page = await backend.listConversationMessages(conversationId, {
    34	    limit: normalizeExtensionConversationHistoryLimit(options?.limit),
    35	    order: "desc",
    36	    include_err: options?.includeErrors ?? true,
    37	    ...(conversationId === "default" && agentId ? { agent_id: agentId } : {}),
    38	  } as ConversationMessageListBody);
    39	  const messages = page.getPaginatedItems() as Message[];
    40	  if (options?.order === "desc") {
    41	    return messages;
    42	  }
    43	  return [...messages].reverse();
    44	}

    18	  name: TName,

    24	  events: ExtensionEvents | undefined,
    25	  name: TName,

    63	export function createExtensionAdapter(
    64	  options: CreateExtensionAdapterOptions,
    65	): ExtensionAdapter {
    66	  const {
    67	    diagnosticsRootDirectory,
    68	    diagnosticsWriteDelayMs = RUNTIME_DIAGNOSTICS_WRITE_DELAY_MS,
    69	    disabled,
    70	    getBackend: resolveBackend,
    71	    initialContext,
    72	    ...engineOptions
    73	  } = options;
    74	
    75	  const alreadyDisabled = areExtensionsDisabled();
    76	  if (disabled || alreadyDisabled) {
    77	    if (!alreadyDisabled) {
    78	      disableExtensionsForProcess();
    79	    }
    80	    return createDisabledExtensionAdapter({ initialContext });
    81	  }
    82	
    83	  let context = initialContext;
    84	  let disposed = false;
    85	  const initialHasExtensionSources = hasExtensionSources(engineOptions);
    86	  let loadState: ExtensionAdapterLoadState = {
    87	    hadStatuslineRenderer: false,
    88	    hasExtensionSources: initialHasExtensionSources,
    89	    isLoading: initialHasExtensionSources,
    90	  };
    91	  const listeners = new Set<() => void>();
    92	  let diagnosticsWriteTimer: ReturnType<typeof setTimeout> | null = null;
    93	
    94	  const getBackend = () => resolveBackend?.();
    95	  const getContext = () => context;
    96	
    97	  const engine = createExtensionEngine({
    98	    ...engineOptions,
    99	    getBackend,
   100	    getContext,
   101	    onDiagnostic: () => scheduleDiagnosticsWrite(),
   102	  });
   103	
   104	  function clearPendingDiagnosticsWrite(): void {
   105	    if (!diagnosticsWriteTimer) return;
   106	    clearTimeout(diagnosticsWriteTimer);
   107	    diagnosticsWriteTimer = null;
   108	  }
   109	
   110	  function writeLatestDiagnostics(): void {
   111	    const registry = engine.getSnapshot();
   112	    if (!registry.sources.some((source) => source.files.length > 0)) return;
   113	
   114	    try {
   115	      writeExtensionDiagnosticsLatestFile(registry.diagnostics, {
   116	        rootDirectory: diagnosticsRootDirectory,
   117	      });
   118	    } catch (error) {
   119	      debugLog(
   120	        "extensions",
   121	        "failed to write extension diagnostics: %s",
   122	        error instanceof Error ? error.message : String(error),
   123	      );
   124	    }
   125	  }
   126	
   127	  function flushPendingDiagnosticsWrite(): void {
   128	    if (!diagnosticsWriteTimer) return;
   129	    clearPendingDiagnosticsWrite();
   130	    writeLatestDiagnostics();
   131	  }
   132	
   133	  function scheduleDiagnosticsWrite(): void {
   134	    if (disposed || loadState.isLoading || !loadState.hasExtensionSources)
   135	      return;
   136	    if (diagnosticsWriteTimer) return;
   137	
   138	    diagnosticsWriteTimer = setTimeout(() => {
   139	      diagnosticsWriteTimer = null;
   140	      writeLatestDiagnostics();
   141	    }, diagnosticsWriteDelayMs);
   142	    const timerWithUnref = diagnosticsWriteTimer as ReturnType<
   143	      typeof setTimeout
   144	    > & { unref?: () => void };
   145	    timerWithUnref.unref?.();
   146	  }
   147	
   148	  const events: ExtensionEvents = {
   149	    async emit(name, event) {
   150	      if (loadState.isLoading || !loadState.hasExtensionSources) {
   151	        // Events are best-effort hooks; do not deliver them while the extension
   152	        // registry is unavailable or in flux.
   153	        return emptyEventEmissionResult(name);
   154	      }
   155	      const result = await engine.emitEvent(name, event);
   156	      return result;
   157	    },
   158	  };
   159	
   160	  const buildSnapshot = (): ExtensionAdapterSnapshot => ({
   161	    registry: engine.getSnapshot(),
   162	    ...loadState,
   163	  });
   164	  let snapshot = buildSnapshot();
   165	
   166	  const publish = () => {
   167	    snapshot = buildSnapshot();
   168	    for (const listener of listeners) {
   169	      listener();
   170	    }
   171	  };
   172	  const unsubscribeEngine = engine.subscribe(publish);
   173	
   174	  const getSnapshot = () => snapshot;
   175	
   176	  const reload = async () => {
   177	    if (disposed) return;
   178	    clearPendingDiagnosticsWrite();
   179	
   180	    const previousSnapshot = engine.getSnapshot();
   181	    const previousHadStatuslineRenderer =
   182	      Boolean(previousSnapshot.ui.statuslineRenderer) ||
   183	      loadState.hadStatuslineRenderer;
   184	    loadState = {
   185	      hadStatuslineRenderer: previousHadStatuslineRenderer,
   186	      hasExtensionSources: hasExtensionSources(engineOptions),
   187	      isLoading: true,
   188	    };
   189	    publish();
   190	
   191	    await engine.reload();
   192	    if (disposed) return;
   193	
   194	    const nextRegistry = engine.getSnapshot();
   195	    writeLatestDiagnostics();
   196	
   197	    debugLog(
   198	      "extensions",
   199	      "loaded %s extension(s) from %s source(s); renderer=%s",
   200	      nextRegistry.loadedPaths.length,
   201	      nextRegistry.sources.length,
   202	      nextRegistry.ui.statuslineRenderer?.id ?? "(none)",
   203	    );
   204	
   205	    for (const diagnostic of getExtensionErrorDiagnostics(
   206	      nextRegistry.diagnostics,
   207	    )) {
   208	      debugLog(
   209	        "extensions",
   210	        "failed to load %s: %s",
   211	        diagnostic.owner.path,
   212	        diagnostic.error.message,
   213	      );
   214	    }
   215	
   216	    for (const diagnostic of nextRegistry.diagnostics) {
   217	      if (diagnostic.phase === "command_override") {
   218	        debugLog("extensions", "%s", diagnostic.error.message);
   219	      }
   220	    }
   221	
   222	    loadState = {
   223	      hadStatuslineRenderer: Boolean(nextRegistry.ui.statuslineRenderer),
   224	      hasExtensionSources: nextRegistry.sources.some(
   225	        (source) => source.files.length > 0,
   226	      ),
   227	      isLoading: false,
   228	    };
   229	    publish();
   230	  };
   231	
   232	  return {
   233	    dispose() {
   234	      if (disposed) return;
   235	      flushPendingDiagnosticsWrite();
   236	      disposed = true;
   237	      engine.dispose();
   238	      unsubscribeEngine();
   239	      listeners.clear();
   240	    },
   241	    events,
   242	    getBackend,
   243	    getContext,
   244	    getSnapshot,
   245	    engine,
   246	    reload,
   247	    subscribe(listener) {
   248	      listeners.add(listener);
   249	      return () => {
   250	        listeners.delete(listener);
   251	      };
   252	    },
   253	    updateContext(nextContext) {
   254	      context = nextContext;
   255	    },
   256	  };
   257	}

   496	  cacheDirectory: string,
   497	): string {

   552	        return getClient().then((client) => {
   553	          let owner: unknown = client;
   554	          let value: unknown = client;
   555	          for (const property of path) {
   556	            owner = value;
   557	            value = (value as Record<PropertyKey, unknown>)[property];
   558	          }
   559	          if (typeof value !== "function") {
   560	            throw new TypeError(
   561	              `letta.client.${path.map(String).join(".")} is not callable`,
   562	            );
   563	          }
   564	          return value.apply(owner, args);
   565	        });
   566	      },
   567	      get(_target, property) {
   568	        // Keep the proxy from being treated as a Promise when code does
   569	        // `await letta.client` or Promise.resolve(letta.client).
   570	        if (property === "then") return undefined;
   571	        return createProxy([...path, property]);
   572	      },
   573	    });
   574	
   575	  return createProxy() as Letta;
   576	}

   829	function createLettaExtensionApi(
   830	  registry: LocalExtensionRegistry,
   831	  owner: ExtensionOwner,
   832	  capabilities: ExtensionCapabilities,
   833	  getClient: () => Promise<Letta>,
   834	  getContext: () => ExtensionContext,
   835	  onChange: () => void,
   836	  onDiagnostic: ((diagnostic: ExtensionDiagnostic) => void) | undefined,
   837	  builtinCommandIds: Set<string>,
   838	  reservedToolNames: Set<string>,
   839	  signal: AbortSignal,
   840	): LettaExtensionApi {
   841	  const isLive = () => isOwnerLive(registry, owner);
   842	  const guardLive = (
   843	    capability: ExtensionDiagnostic["capability"],
   844	  ): boolean => {
   845	    if (isLive()) return true;
   846	    recordStaleHandleUse(registry, owner, capability, onDiagnostic);
   847	    return false;
   848	  };
   849	
   850	  const unregisterEvent = <TName extends ExtensionEventName>(
   851	    name: TName,
   852	    handler: ExtensionEventHandler<TName>,
   853	  ) => {
   854	    validateExtensionEventName(name);
   855	    if (!isExtensionEventCapabilityEnabled(capabilities, name)) return;
   856	    if (!guardLive({ id: name, kind: "event" })) return;
   857	    const registrations = registry.events[name];
   858	    if (!registrations) return;
   859	    const nextRegistrations = registrations.filter(
   860	      (registration) =>
   861	        registration.owner?.id !== owner.id ||
   862	        registration.handler !== (handler as unknown as ExtensionEventHandler),
   863	    );
   864	    if (nextRegistrations.length > 0) {
   865	      registry.events[name] = nextRegistrations;
   866	    } else {
   867	      delete registry.events[name];
   868	    }
   869	    onChange();
   870	  };
   871	
   872	  const unregisterCommand = (id: string) => {
   873	    if (!capabilities.commands) return;
   874	    validateExtensionCommandId(id);
   875	    if (!guardLive({ id, kind: "command" })) return;
   876	    const existing = registry.commands[id];
   877	    if (existing?.owner?.id === owner.id) {
   878	      delete registry.commands[id];
   879	      onChange();
   880	    }
   881	  };
   882	
   883	  const unregisterPermission = (id: string) => {
   884	    if (!capabilities.permissions) return;
   885	    validateExtensionPermissionId(id);
   886	    if (!guardLive({ id, kind: "permission" })) return;
   887	    const existing = registry.permissions[id];
   888	    if (existing?.owner?.id === owner.id) {
   889	      delete registry.permissions[id];
   890	      unregisterExtensionPermission(id, owner);
   891	      onChange();
   892	    }
   893	  };
   894	
   895	  const clearPanel = (id: string) => {
   896	    if (!capabilities.ui.panels) return;
   897	    validateExtensionPanelId(id);
   898	    if (!guardLive({ id, kind: "panel" })) return;
   899	    const panelKey = getExtensionPanelKey(owner.id, id);
   900	    const existing = registry.ui.panels[panelKey];
   901	    if (existing?.owner?.id === owner.id) {
   902	      delete registry.ui.panels[panelKey];
   903	      onChange();
   904	    }
   905	  };
   906	
   907	  const unregisterTool = (name: string) => {
   908	    if (!capabilities.tools) return;
   909	    validateExtensionToolName(name);
   910	    if (!guardLive({ id: name, kind: "tool" })) return;
   911	    const existing = registry.tools[name];
   912	    if (existing?.owner?.id === owner.id) {
   913	      delete registry.tools[name];
   914	      unregisterExtensionTool(name, owner);
   915	      onChange();
   916	    }
   917	  };
   918	
   919	  const unregisterProvider = (name: string) => {
   920	    if (!capabilities.providers) return;
   921	    if (!guardLive({ id: name, kind: "provider" })) return;
   922	    unregisterPiProvider(name, owner.id);
   923	    clearAvailableModelsCache();
   924	    onChange();
   925	  };
   926	
   927	  const registerProviderForOwner = (
   928	    name: string,
   929	    config: PiProviderRegistration,
   930	  ): LettaExtensionDisposer => {
   931	    if (!capabilities.providers) {
   932	      return () => undefined;
   933	    }
   934	    if (!guardLive({ id: name, kind: "provider" })) {
   935	      return () => undefined;
   936	    }
   937	    registerPiProvider(name, config, {
   938	      id: owner.id,
   939	      path: owner.path,
   940	    });
   941	    clearAvailableModelsCache();
   942	    onChange();
   943	    return () => unregisterProvider(name);
   944	  };
   945	
   946	  const normalizeReportedDiagnostic = (
   947	    diagnostic: ExtensionDiagnosticReportOptions,
   948	  ): { message: string; severity: ExtensionDiagnosticSeverity } => {
   949	    if (!diagnostic || typeof diagnostic !== "object") {
   950	      throw new Error("Extension diagnostic report must be an object");
   951	    }
   952	    if (typeof diagnostic.message !== "string") {
   953	      throw new Error("Extension diagnostic report must include a message");
   954	    }
   955	    const message = diagnostic.message.trim();
   956	    if (message.length === 0) {
   957	      throw new Error("Extension diagnostic report message cannot be empty");
   958	    }
   959	    if (
   960	      diagnostic.severity !== undefined &&
   961	      diagnostic.severity !== "error" &&
   962	      diagnostic.severity !== "warning"
   963	    ) {
   964	      throw new Error(
   965	        "Extension diagnostic severity must be 'error' or 'warning'",
   966	      );
   967	    }
   968	    return { message, severity: diagnostic.severity ?? "error" };
   969	  };
   970	
   971	  const reportDiagnostic = (diagnostic: ExtensionDiagnosticReportOptions) => {
   972	    if (!guardLive(undefined)) return;
   973	    const normalized = normalizeReportedDiagnostic(diagnostic);
   974	    const error = new Error(normalized.message);
   975	    error.name = "ExtensionDiagnosticReport";
   976	    error.stack = undefined;
   977	    recordExtensionDiagnostic(
   978	      registry,
   979	      {
   980	        error,
   981	        owner,
   982	        phase: "report",
   983	        severity: normalized.severity,
   984	      },
   985	      onDiagnostic,
   986	    );
   987	  };
   988	
   989	  const onEvent = <TName extends ExtensionEventName>(
   990	    name: TName,
   991	    handler: ExtensionEventHandler<TName>,
   992	  ): LettaExtensionDisposer => {
   993	    validateExtensionEventName(name);
   994	    if (!isExtensionEventCapabilityEnabled(capabilities, name)) {
   995	      return () => undefined;
   996	    }
   997	    if (typeof handler !== "function") {
   998	      throw new Error("Extension event registration must include a handler");
   999	    }
  1000	    if (!guardLive({ id: name, kind: "event" })) {
  1001	      return () => undefined;
  1002	    }
  1003	
  1004	    registry.events[name] = [
  1005	      ...(registry.events[name] ?? []),
  1006	      {
  1007	        handler: handler as unknown as ExtensionEventHandler,
  1008	        name,
  1009	        owner,
  1010	      },
  1011	    ];
  1012	    onChange();
  1013	
  1014	    return () => unregisterEvent(name, handler);
  1015	  };
  1016	
  1017	  return {
  1018	    capabilities: cloneExtensionCapabilities(capabilities),
  1019	    client: createLazyClient(getClient),
  1020	    getClient,
  1021	    getContext,
  1022	    signal,
  1023	    registerProvider: registerProviderForOwner,
  1024	    unregisterProvider,
  1025	    commands: {
  1026	      register(command) {
  1027	        if (!capabilities.commands) {
  1028	          return () => undefined;
  1029	        }
  1030	        if (!guardLive({ id: command.id, kind: "command" })) {
  1031	          return () => undefined;
  1032	        }
  1033	
  1034	        const normalized = normalizeExtensionCommand(command, owner);
  1035	        if (builtinCommandIds.has(normalized.id)) {
  1036	          recordExtensionDiagnostic(
  1037	            registry,
  1038	            {
  1039	              capability: { id: normalized.id, kind: "command" },
  1040	              error: new Error(
  1041	                `Extension command '${normalized.id}' overrides a built-in command`,
  1042	              ),
  1043	              owner,
  1044	              phase: "command_override",
  1045	            },
  1046	            onDiagnostic,
  1047	          );
  1048	        }
  1049	
  1050	        const existing = registry.commands[normalized.id];
  1051	        if (existing && !command.override) {
  1052	          throw new Error(
  1053	            `Extension command '${normalized.id}' is already registered by ${existing.path}`,
  1054	          );
  1055	        }
  1056	
  1057	        registry.commands[normalized.id] = normalized;
  1058	        onChange();
  1059	
  1060	        return () => unregisterCommand(normalized.id);
  1061	      },
  1062	      unregister: unregisterCommand,
  1063	    },
  1064	    tools: {
  1065	      register(tool) {
  1066	        if (!capabilities.tools) {
  1067	          return () => undefined;
  1068	        }
  1069	        if (!guardLive({ id: tool.name, kind: "tool" })) {
  1070	          return () => undefined;
  1071	        }
  1072	
  1073	        const normalized = normalizeExtensionTool(tool, owner);
  1074	        if (reservedToolNames.has(normalized.name)) {
  1075	          throw new Error(
  1076	            `Extension tool '${normalized.name}' conflicts with a built-in tool`,
  1077	          );
  1078	        }
  1079	
  1080	        const existing = registry.tools[normalized.name];
  1081	        const existingGlobal = getExtensionToolDefinition(normalized.name);
  1082	        if ((existing || existingGlobal) && !tool.override) {
  1083	          throw new Error(
  1084	            `Extension tool '${normalized.name}' is already registered by ${existing?.path ?? existingGlobal?.path}`,
  1085	          );
  1086	        }
  1087	
  1088	        registry.tools[normalized.name] = normalized;
  1089	        registerExtensionTool({
  1090	          ...normalized,
  1091	          activationSignal: signal,
  1092	          getContext,
  1093	          isAvailable: () => {
  1094	            if (signal.aborted) return false;
  1095	            return normalized.isEnabled?.(getContext()) ?? true;
  1096	          },
  1097	        });
  1098	        onChange();
  1099	
  1100	        return () => unregisterTool(normalized.name);
  1101	      },
  1102	      unregister: unregisterTool,
  1103	    },
  1104	    providers: {
  1105	      register(name, config) {
  1106	        return registerProviderForOwner(name, config);
  1107	      },
  1108	      unregister: unregisterProvider,
  1109	    },
  1110	    events: {
  1111	      off: unregisterEvent,
  1112	      on: onEvent,
  1113	    },
  1114	    permissions: {
  1115	      register(permission) {
  1116	        if (!capabilities.permissions) {
  1117	          return () => undefined;
  1118	        }
  1119	        if (!guardLive({ id: permission.id, kind: "permission" })) {
  1120	          return () => undefined;
  1121	        }
  1122	
  1123	        const normalized = normalizeExtensionPermission(permission, owner);
  1124	        const existing = registry.permissions[normalized.id];
  1125	        const existingGlobal = getExtensionPermissionDefinition(normalized.id);
  1126	        if (existing || existingGlobal) {
  1127	          throw new Error(
  1128	            `Extension permission '${normalized.id}' is already registered by ${existing?.path ?? existingGlobal?.path}`,
  1129	          );
  1130	        }
  1131	
  1132	        registry.permissions[normalized.id] = normalized;
  1133	        registerExtensionPermission({
  1134	          ...normalized,
  1135	          activationSignal: signal,
  1136	          getContext,
  1137	          isAvailable: () => {
  1138	            if (signal.aborted) return false;
  1139	            return normalized.isEnabled?.(getContext()) ?? true;
  1140	          },
  1141	        });
  1142	        onChange();
  1143	
  1144	        return () => unregisterPermission(normalized.id);
  1145	      },
  1146	      unregister: unregisterPermission,
  1147	    },
  1148	    diagnostics: {
  1149	      report: reportDiagnostic,
  1150	    },
  1151	    ui: {
  1152	      clearPanel,
  1153	      clearStatus(key) {
  1154	        if (!capabilities.ui.statusValues) return;
  1155	        if (!guardLive({ id: key, kind: "status" })) return;
  1156	        delete registry.ui.statusValues[key];
  1157	        delete registry.ui.statusOwners[key];
  1158	        onChange();
  1159	      },
  1160	      openPanel(panel) {
  1161	        if (!capabilities.ui.panels) {
  1162	          return {
  1163	            close() {},
  1164	            update() {},
  1165	          };
  1166	        }
  1167	        if (!guardLive({ id: panel.id, kind: "panel" })) {
  1168	          return {
  1169	            close() {},
  1170	            update() {},
  1171	          };
  1172	        }
  1173	
  1174	        upsertExtensionPanel(registry, owner, panel.id, panel);
  1175	        onChange();
  1176	        return {
  1177	          close() {
  1178	            clearPanel(panel.id);
  1179	          },
  1180	          update(update) {
  1181	            if (!guardLive({ id: panel.id, kind: "panel" })) return;
  1182	            upsertExtensionPanel(registry, owner, panel.id, update);
  1183	            onChange();
  1184	          },
  1185	        };
  1186	      },
  1187	      setStatus(key, value) {
  1188	        if (!capabilities.ui.statusValues) return;
  1189	        if (!guardLive({ id: key, kind: "status" })) return;
  1190	        if (value == null) {
  1191	          delete registry.ui.statusValues[key];
  1192	          delete registry.ui.statusOwners[key];
  1193	          onChange();
  1194	          return;
  1195	        }
  1196	        registry.ui.statusValues[key] = value;
  1197	        registry.ui.statusOwners[key] = owner;
  1198	        onChange();

  1220	export async function loadLocalExtensions(
  1221	  options: LoadLocalExtensionsOptions,
  1222	): Promise<LocalExtensionRegistry> {

  1225	  const getConfiguredClient = () => {
  1226	    clientPromise ??= options.getClient();

  1228	  };
  1229	  const getContext =
  1230	    options.getContext ??
  1231	    (() => {

  1233	    });

  1235	  const sources = resolveLocalExtensionSources(options);
  1236	  const capabilities = resolveExtensionCapabilities(options.capabilities);
  1237	  const generation = options.generation ?? 1;
  1238	  const builtinCommandIds = new Set([...(options.builtinCommandIds ?? [])]);
  1239	  const reservedToolNames = new Set([...(options.reservedToolNames ?? [])]);
  1240	  const registry = createEmptyExtensionRegistry(
  1241	    sources,
  1242	    generation,
  1243	    capabilities,
  1244	  );

  1248	      const owner = createExtensionOwner(extensionPath, source, generation);

  1251	      registry.ownerAbortControllers[owner.id] = abortController;
  1252	      registry.owners[owner.id] = owner;

  1255	        const mtimeMs = statSync(extensionPath).mtimeMs;
  1256	        failurePhase = TYPESCRIPT_EXTENSION_FILE_EXTENSIONS.has(

  1270	        failurePhase = "activate";

  1275	          );

  1279	          createLettaExtensionApi(

  1283	            getConfiguredClient,
  1284	            getContext,

  1287	            builtinCommandIds,
  1288	            reservedToolNames,
  1289	            abortController.signal,
  1290	          ),
  1291	        );

  1297	          });

  1300	      } catch (error) {

  1312	        );
  1313	      }

  1321	  registry: LocalExtensionRegistry | null,
  1322	  context: ExtensionContext,
  1323	): Record<string, string> {

  1342	export async function emitLocalExtensionEvent<TName extends ExtensionEventName>(
  1343	  registry: LocalExtensionRegistry | null,
  1344	  name: TName,
  1345	  event: ExtensionEventMap[TName],
  1346	  getContext: () => ExtensionContext,
  1347	  backend?: Backend,
  1348	  onDiagnostic?: (diagnostic: ExtensionDiagnostic) => void,
  1349	): Promise<ExtensionEventEmissionResult<TName>> {
  1350	  if (!registry) {
  1351	    return { diagnostics: [], handlerCount: 0, name, results: [] };
  1352	  }
  1353	
  1354	  validateExtensionEventName(name);
  1355	  const registrations = [...(registry.events[name] ?? [])];
  1356	  const diagnostics: ExtensionDiagnostic[] = [];
  1357	  const results: Array<NonNullable<ExtensionEventResultMap[TName]>> = [];
  1358	
  1359	  for (const registration of registrations) {
  1360	    const signal = registration.owner
  1361	      ? registry.ownerAbortControllers[registration.owner.id]?.signal
  1362	      : undefined;
  1363	    if (signal?.aborted) continue;
  1364	    const turnStartEvent =
  1365	      name === "turn_start" ? (event as ExtensionTurnStartEvent) : null;
  1366	    const turnStartInputBeforeHandler =
  1367	      turnStartEvent && isTurnStartInput(turnStartEvent.input)
  1368	        ? cloneTurnStartInput(turnStartEvent.input)
  1369	        : null;
  1370	    const toolStartEvent =
  1371	      name === "tool_start" ? (event as ExtensionToolStartEvent) : null;
  1372	    const toolStartArgsBeforeHandler =
  1373	      toolStartEvent && isToolStartArgs(toolStartEvent.args)
  1374	        ? cloneToolStartArgs(toolStartEvent.args)
  1375	        : null;
  1376	
  1377	    try {
  1378	      const context = getContext();
  1379	      const eventContext: ExtensionEventContext = {
  1380	        conversation: createExtensionConversationHandle({
  1381	          agentId:
  1382	            typeof event.agentId === "string"
  1383	              ? event.agentId
  1384	              : context.agent.id,
  1385	          backend,
  1386	          conversationId:
  1387	            typeof event.conversationId === "string"
  1388	              ? event.conversationId
  1389	              : context.sessionId,
  1390	          sendMessageStream: sendMessageStreamWithBackend,
  1391	          workingDirectory: context.cwd,
  1392	        }),
  1393	        context,
  1394	        getContext,
  1395	        signal: signal ?? new AbortController().signal,
  1396	      };
  1397	      const result = await registration.handler(event, eventContext);
  1398	      if (isTurnStartResultWithInput(name, result)) {
  1399	        (event as ExtensionTurnStartEvent).input = result.input;
  1400	      }
  1401	      if (isToolStartResultWithArgs(name, result)) {
  1402	        (event as ExtensionToolStartEvent).args = result.args;
  1403	      }
  1404	      if (result != null) {
  1405	        results.push(result as NonNullable<ExtensionEventResultMap[TName]>);
  1406	      }
  1407	      if (
  1408	        turnStartEvent &&
  1409	        turnStartInputBeforeHandler &&
  1410	        !isTurnStartInput(turnStartEvent.input)
  1411	      ) {
  1412	        turnStartEvent.input = turnStartInputBeforeHandler;
  1413	      }
  1414	      if (
  1415	        toolStartEvent &&
  1416	        toolStartArgsBeforeHandler &&
  1417	        !isToolStartArgs(toolStartEvent.args)
  1418	      ) {
  1419	        toolStartEvent.args = toolStartArgsBeforeHandler;
  1420	      }
  1421	    } catch (error) {
  1422	      if (turnStartEvent && turnStartInputBeforeHandler) {
  1423	        turnStartEvent.input = turnStartInputBeforeHandler;
  1424	      }
  1425	      if (toolStartEvent && toolStartArgsBeforeHandler) {
  1426	        toolStartEvent.args = toolStartArgsBeforeHandler;
  1427	      }
  1428	      const diagnostic = recordExtensionDiagnostic(
  1429	        registry,
  1430	        {
  1431	          capability: { id: name, kind: "event" },
  1432	          error: error instanceof Error ? error : new Error(String(error)),
  1433	          owner: registration.owner,
  1434	          phase: "event",
  1435	        },
  1436	        onDiagnostic,
  1437	      );
  1438	      diagnostics.push(diagnostic);
  1439	    }
  1440	  }
  1441	
  1442	  return { diagnostics, handlerCount: registrations.length, name, results };
  1443	}

  1485	export function createExtensionEngine(
  1486	  options: CreateExtensionEngineOptions,
  1487	): ExtensionEngine {
  1488	  const { getBackend, onDiagnostic, ...extensionOptions } = options;
  1489	  let generation = 0;
  1490	  let disposed = false;
  1491	  const capabilities = resolveExtensionCapabilities(
  1492	    extensionOptions.capabilities,
  1493	  );
  1494	  const getContext =
  1495	    extensionOptions.getContext ??
  1496	    (() => {
  1497	      throw new Error("Extension context is not available yet");
  1498	    });
  1499	  let activeRegistry = createEmptyExtensionRegistry(
  1500	    resolveLocalExtensionSources(extensionOptions),
  1501	    generation,
  1502	    capabilities,
  1503	  );
  1504	  let snapshot = snapshotRegistryForReaders(activeRegistry);
  1505	  const listeners = new Set<() => void>();
  1506	
  1507	  const publish = () => {
  1508	    snapshot = snapshotRegistryForReaders(activeRegistry);
  1509	    for (const listener of listeners) {
  1510	      listener();
  1511	    }
  1512	  };
  1513	
  1514	  const reload = async () => {
  1515	    if (disposed) return;
  1516	
  1517	    disposeLocalExtensions(activeRegistry);
  1518	    generation += 1;
  1519	    const loadGeneration = generation;
  1520	    activeRegistry = createEmptyExtensionRegistry(
  1521	      resolveLocalExtensionSources(extensionOptions),
  1522	      loadGeneration,
  1523	      capabilities,
  1524	    );
  1525	    publish();
  1526	
  1527	    let loadingRegistry: LocalExtensionRegistry | null = null;
  1528	    const nextRegistry = await loadLocalExtensions({
  1529	      ...extensionOptions,
  1530	      generation: loadGeneration,
  1531	      onChange: () => {
  1532	        if (!disposed && loadingRegistry && loadGeneration === generation) {
  1533	          activeRegistry = loadingRegistry;
  1534	          publish();
  1535	        }
  1536	      },
  1537	      onDiagnostic: (diagnostic) => {
  1538	        if (disposed) return;
  1539	
  1540	        if (loadingRegistry && loadGeneration === generation) {
  1541	          activeRegistry = loadingRegistry;
  1542	          publish();
  1543	          onDiagnostic?.(diagnostic);
  1544	          return;
  1545	        }
  1546	        // Stale handles from a prior generation report through their old
  1547	        // activation callback. Preserve the diagnostic on the current engine
  1548	        // snapshot without reviving the old registry.
  1549	        appendExtensionDiagnostic(activeRegistry, diagnostic);
  1550	        publish();
  1551	        onDiagnostic?.(diagnostic);
  1552	      },
  1553	    });
  1554	    loadingRegistry = nextRegistry;
  1555	    if (disposed || loadGeneration !== generation) {
  1556	      disposeLocalExtensions(nextRegistry);
  1557	      return;
  1558	    }
  1559	
  1560	    activeRegistry = nextRegistry;
  1561	    publish();
  1562	  };
  1563	
  1564	  return {
  1565	    dispose() {
  1566	      if (disposed) return;
  1567	      disposed = true;
  1568	      generation += 1;
  1569	      disposeLocalExtensions(activeRegistry);
  1570	      activeRegistry = createEmptyExtensionRegistry(
  1571	        resolveLocalExtensionSources(extensionOptions),
  1572	        generation,
  1573	        capabilities,
  1574	      );
  1575	      publish();
  1576	      listeners.clear();
  1577	    },
  1578	    async emitEvent(name, payload) {
  1579	      if (disposed) {
  1580	        return { diagnostics: [], handlerCount: 0, name, results: [] };
  1581	      }
  1582	      const invocationBackend = getBackend?.();
  1583	      const result = await emitLocalExtensionEvent(
  1584	        activeRegistry,
  1585	        name,
  1586	        payload,
  1587	        getContext,
  1588	        invocationBackend,
  1589	        onDiagnostic,
  1590	      );
  1591	      if (result.diagnostics.length > 0) {
  1592	        publish();
  1593	      }
  1594	      return result;
  1595	    },
  1596	    getSnapshot() {
  1597	      return snapshot;
  1598	    },
  1599	    reload,
  1600	    subscribe(listener) {
  1601	      listeners.add(listener);
  1602	      return () => {
  1603	        listeners.delete(listener);
  1604	      };
  1605	    },
  1606	  };
  1607	}
